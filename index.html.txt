        function renderProducts(filter = 'all', search = '') {
            const grid = document.getElementById('productGrid');

        // Filter Handling
        function filterCategory(category) {
            currentFilter = category;
            document.querySelectorAll('.cat-btn').forEach(btn => {
                btn.classList.remove('bg-neoncyan', 'text-black', 'shadow-md');
                btn.classList.add('bg-cardbg', 'border', 'border-gray-800', 'text-gray-300');
            });
            event.target.classList.remove('bg-cardbg', 'border', 'border-gray-800', 'text-gray-300');
            event.target.classList.add('bg-neoncyan', 'text-black', 'shadow-md');
            
            const searchVal = document.getElementById('searchInput').value;
            renderProducts(category, searchVal);
        }

        function handleSearch() {
            const searchVal = document.getElementById('searchInput').value;
            renderProducts(currentFilter, searchVal);
        }

        // Cart Logic
        function toggleCart() {
            const drawer = document.getElementById('cartDrawer');
            if (drawer.classList.contains('hidden')) {
                drawer.classList.remove('hidden');
                drawer.classList.add('flex');
            } else {
                drawer.classList.remove('flex');
                drawer.classList.add('hidden');
            }
        }

        function addToCart(id) {
            const product = products.find(p => p.id === id);
            const existing = cart.find(item => item.id === id);
            if (existing) {
                existing.qty++;
            } else {
                cart.push({ ...product, qty: 1 });
            }
            updateCartUI();
            toggleCart();
        }

        function updateCartUI() {
            const countEl = document.getElementById('cartCount');
            const listEl = document.getElementById('cartItemsList');
            const totalEl = document.getElementById('cartTotal');

            const totalCount = cart.reduce((sum, item) => sum + item.qty, 0);
            countEl.innerText = totalCount;

            if (cart.length === 0) {
                listEl.innerHTML = `<div class="text-center py-12 text-gray-500 font-orbitron text-xs">Votre panier est vide.</div>`;
                totalEl.innerText = "0 MAD";
                return;
            }

            listEl.innerHTML = '';
            let totalPrice = 0;

            cart.forEach(item => {
                totalPrice += item.price * item.qty;
                listEl.innerHTML += `
                    <div class="flex items-center justify-between bg-darkbg p-3 rounded-xl border border-gray-800">
                        <div class="flex items-center space-x-3">
                            <div class="w-10 h-10 bg-cardbg rounded-lg border border-gray-800 flex items-center justify-center text-neoncyan">
                                <i class="fa-solid ${item.image}"></i>
                            </div>
                            <div>
                                <h4 class="font-orbitron font-bold text-white text-xs">${item.name}</h4>
                                <span class="text-neoncyan text-xs font-bold">${item.price} MAD</span>
                            </div>
                        </div>
                        <div class="flex items-center space-x-2">
                            <button onclick="changeQty(${item.id}, -1)" class="w-6 h-6 bg-cardbg border border-gray-800 rounded text-white flex items-center justify-center text-xs">-</button>
                            <span class="text-xs font-bold text-white">${item.qty}</span>
                            <button onclick="changeQty(${item.id}, 1)" class="w-6 h-6 bg-cardbg border border-gray-800 rounded text-white flex items-center justify-center text-xs">+</button>
                        </div>
                    </div>
                `;
            });

            totalEl.innerText = `${totalPrice} MAD`;
        }

        function changeQty(id, delta) {
            const item = cart.find(i => i.id === id);
            if (item) {
                item.qty += delta;
                if (item.qty <= 0) {
                    cart = cart.filter(i => i.id !== id);
                }
            }
            updateCartUI();
        }

        // WhatsApp Checkout
        function checkoutWhatsApp() {
            if (cart.length === 0) {
                alert("Votre panier est vide !");
                return;
            }

            const name = document.getElementById('clientName').value.trim();
            const phone = document.getElementById('clientPhone').value.trim();
            const city = document.getElementById('clientCity').value.trim();

            if (!name || !phone || !city) {
                alert("Veuillez remplir toutes vos informations de livraison (Nom, Téléphone, Ville).");
                return;
            }

            let message = `🛒 *NOUVELLE COMMANDE - NEXUS GEAR*%0A%0A`;
            message += `👤 *Nom:* ${name}%0A`;
            message += `📞 *Téléphone:* ${phone}%0A`;
            message += `🏙️ *Ville:* ${city}%0A%0A`;
            message += `📦 *Produits:*%0A`;

            let total = 0;
            cart.forEach(item => {
                let sub = item.price * item.qty;
                total += sub;
                message += `- ${item.name} (x${item.qty}) : ${sub} MAD%0A`;
            });

            message += `%0A💰 *TOTAL:* *${total} MAD*%0A`;
            message += `🚚 *Paiement à la livraison (COD)*`;

            // Numéro WhatsApp (Baddlo b numro dyalk)
            const whatsappNumber = "212600000000"; 
            const url = `https://wa.me/${whatsappNumber}?text=${message}`;
            window.open(url, '_blank');
        }
